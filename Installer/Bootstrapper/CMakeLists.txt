cmake_minimum_required(VERSION 3.20)
project(GamEngineSetup LANGUAGES CXX RC)
option(GE_SETUP_ENABLE_TEST_COMMANDS "Enable internal install/uninstall test commands" OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(GamEngineSetup WIN32 main.cpp ReleaseClient.cpp setup.rc)
target_compile_features(GamEngineSetup PRIVATE cxx_std_20)
target_compile_definitions(GamEngineSetup PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(GE_SETUP_ENABLE_TEST_COMMANDS)
    target_compile_definitions(GamEngineSetup PRIVATE GE_SETUP_TEST_COMMANDS=1)
endif()
target_link_libraries(GamEngineSetup PRIVATE winhttp bcrypt shell32 ole32 comctl32 advapi32 dwmapi gdiplus)
if(MSVC)
    target_link_options(GamEngineSetup PRIVATE /MANIFEST:NO)
endif()
